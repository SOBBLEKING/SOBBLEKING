-- SOBBLEKING：訂單編號 SB- 前綴統一改為 SK- -- 先在 Supabase SQL Editor 執行。此段依照目前 schema.sql 的 public.orders / create_customer_order 架構撰寫。 begin; -- 1. 將既有訂單編號 SB-xxxx 改為 SK-xxxx -- 若已有相同 SK 編號，唯一限制會阻止更新並整筆回滾，避免覆蓋資料。 update public.orders set order_code = 'SK-' || substring(order_code from 4) where order_code like 'SB-%'; -- 2. 新訂單一律產生 SK- 前綴 create or replace function public.create_customer_order( p_product_id text, p_quantity integer, p_login_method text, p_game_account text, p_payment_method text, p_notes text default '' ) returns json language plpgsql security definer set search_path=public as $$ declare v_product public.products%rowtype; v_fee integer; v_code text; v_token text := encode(gen_random_bytes(24),'hex'); v_id uuid; v_attempt integer := 0; v_qty integer := greatest(1,least(coalesce(p_quantity,1),99)); begin if p_login_method not in ('PTC','Google','Facebook') then raise exception '登入方式無效'; end if; if length(trim(coalesce(p_game_account,''))) < 2 then raise exception '請填寫遊戲帳號'; end if; if length(trim(coalesce(p_payment_method,''))) < 1 then raise exception '請選擇付款方式'; end if; select * into v_product from public.products where id=p_product_id and active=true; if not found then raise exception '商品不存在或已下架'; end if; v_fee := case when p_login_method='PTC' then 0 else 50 end; loop v_code := 'SK-' || upper(substr(encode(gen_random_bytes(8),'hex'),1,8)); v_attempt := v_attempt + 1; exit when not exists(select 1 from public.orders where order_code=v_code); if v_attempt > 10 then raise exception '訂單編號產生失敗，請重試'; end if; end loop; insert into public.orders(order_code,lookup_token_hash,product_id,product_name,quantity,login_method,game_account,subtotal,login_fee,total_amount,payment_method,notes) values(v_code,encode(digest(v_token,'sha256'),'hex'),v_product.id,v_product.name,v_qty,p_login_method,trim(p_game_account),v_product.price*v_qty,v_fee*v_qty,(v_product.price+v_fee)*v_qty,trim(p_payment_method),left(coalesce(p_notes,''),500)) returning id into v_id; return json_build_object('order_code',v_code,'lookup_token',v_token,'total_amount',(v_product.price+v_fee)*v_qty); end $$; revoke all on function public.create_customer_order(text,integer,text,text,text,text) from public; grant execute on function public.create_customer_order(text,integer,text,text,text,text) to anon, authenticated; commit; -- 3. 查詢會員資料表與會員編號欄位，結果用來精準更新既有會員編號及會員編號產生邏輯。 -- 目前已找到 line-member-login Edge Function 的呼叫端，但尚未取得函式原始碼與會員表實際名稱，故不猜表名、不執行不安全的全庫替換。 select table_schema, table_name, column_name, data_type from information_schema.columns where table_schema = 'public' and (column_name ilike '%member%' or column_name ilike '%line%' or column_name ilike '%no%') order by table_name, ordinal_position;
